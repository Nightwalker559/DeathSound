@echo off
setlocal EnableDelayedExpansion

echo.
echo  ==========================================
echo   DeathSound - List Generator
echo  ==========================================
echo.

set "ADDON_DIR=%~dp0"
if "%ADDON_DIR:~-1%"=="\" set "ADDON_DIR=%ADDON_DIR:~0,-1%"

set "SOUNDS_DIR=%ADDON_DIR%\Media\Sounds"
set "IMAGES_DIR=%ADDON_DIR%\Media\Images"
set "OUT=%ADDON_DIR%\Data\MediaLists.lua"

echo  Addon-Ordner  : %ADDON_DIR%
echo  Sounds-Ordner : %SOUNDS_DIR%
echo  Images-Ordner : %IMAGES_DIR%
echo.

set ERRORS=0
set SOUND_COUNT=0
set IMAGE_COUNT=0

:: Both lists are always written to one file. If a folder is missing or empty,
:: the matching list is written empty (otherwise a stale list would stay behind).

> "%OUT%" echo -- ============================================================
>> "%OUT%" echo -- DeathSound - MediaLists.lua
>> "%OUT%" echo -- AUTO-GENERATED by UpdateLists.bat
>> "%OUT%" echo -- Generated on: %DATE% at %TIME%
>> "%OUT%" echo -- ============================================================
>> "%OUT%" echo.
>> "%OUT%" echo local _, ns = ...
>> "%OUT%" echo.

:: ==========================================
:: SOUNDS
:: ==========================================

echo  --- Sounds ---
echo.

>> "%OUT%" echo -- Sounds (.ogg, .mp3)
>> "%OUT%" echo ns.SoundList = {

if not exist "%SOUNDS_DIR%\" (
    echo  [FEHLER] Sounds-Ordner nicht gefunden!
    echo  Erwartet: %SOUNDS_DIR%
    echo.
    set /a ERRORS+=1
) else (
    for %%F in ("%SOUNDS_DIR%\*.ogg" "%SOUNDS_DIR%\*.mp3") do (
        set "FNAME=%%~nxF"
        set "DNAME=%%~nF"
        set "DNAME=!DNAME:_= !"
        set "DNAME=!DNAME:-= !"
        >> "%OUT%" echo     { name = "!DNAME!", file = "!FNAME!" },
        echo   [+] %%~nxF
        set /a SOUND_COUNT+=1
    )
    if !SOUND_COUNT!==0 (
        echo  [HINWEIS] Keine .ogg oder .mp3 Dateien gefunden.
        echo  Lege zuerst Sounddateien in: %SOUNDS_DIR%
    )
    echo.
)

>> "%OUT%" echo }
>> "%OUT%" echo.

:: ==========================================
:: IMAGES
:: ==========================================

echo  --- Emote-Bilder ---
echo.

>> "%OUT%" echo -- Images
>> "%OUT%" echo -- Recommended format: .png (small, alpha channel, universal)
>> "%OUT%" echo -- Legacy format:      .tga (still supported)
>> "%OUT%" echo -- Max. size in game: 128x128 px
>> "%OUT%" echo ns.ImageList = {

if not exist "%IMAGES_DIR%\" (
    echo  [FEHLER] Images-Ordner nicht gefunden!
    echo  Erwartet: %IMAGES_DIR%
    echo.
    set /a ERRORS+=1
) else (
    for %%F in ("%IMAGES_DIR%\*.png" "%IMAGES_DIR%\*.tga") do (
        set "FNAME=%%~nxF"
        set "DNAME=%%~nF"
        set "DNAME=!DNAME:_= !"
        set "DNAME=!DNAME:-= !"
        >> "%OUT%" echo     { name = "!DNAME!", file = "!FNAME!" },
        echo   [+] %%~nxF
        set /a IMAGE_COUNT+=1
    )
    if !IMAGE_COUNT!==0 (
        echo  [HINWEIS] Keine .png oder .tga Dateien gefunden.
        echo  Lege zuerst Bilddateien in: %IMAGES_DIR%
    )
    echo.
)

>> "%OUT%" echo }

:: ==========================================
:: SUMMARY
:: ==========================================

echo  [OK] MediaLists.lua geschrieben. (%SOUND_COUNT% Sound(e), %IMAGE_COUNT% Bild(er))
echo.
echo  ==========================================
if %ERRORS%==0 (
    echo  Alle Listen erfolgreich aktualisiert.
) else (
    echo  [!] %ERRORS% Fehler aufgetreten - siehe oben.
)
echo.
echo  WICHTIG: Danach im Spiel /reload ausfuehren!
echo  (ein kompletter Neustart ist nicht noetig)
echo  ==========================================

echo.
pause
